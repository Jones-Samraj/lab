import pandas as pd

def covers(h, x):
    return all(a == '?' or a == b for a, b in zip(h, x))

def find_s(data):
    h = None
    for *x, label in data:
        if label == "Yes":
            h = x[:] if h is None else [a if a == b else '?' for a, b in zip(h, x)]
    return h

def candidate_elimination(data):
    n = len(data[0]) - 1
    domains = [sorted({r[i] for r in data}) for i in range(n)]
    S, G = None, [['?'] * n]
    for *x, label in data:
        if label == "Yes":
            G = [g for g in G if covers(g, x)]
            S = x[:] if S is None else [a if a == b else '?' for a, b in zip(S, x)]
        else:
            newG = []
            for g in G:
                if not covers(g, x):
                    newG.append(g); continue
                for i in range(n):
                    if g[i] == '?':
                        for v in domains[i]:
                            if v != x[i]:
                                h = g[:]; h[i] = v
                                if S is None or covers(h, S):
                                    newG.append(h)
            G = newG
    return S, G

import kagglehub
import numpy as np
import os
import cv2
from tqdm import tqdm
from sklearn.svm import SVC
from sklearn.model_selection import train_test_split
from sklearn.preprocessing import StandardScaler
from sklearn.metrics import accuracy_score, classification_report
from sklearn.decomposition import PCA
from multiprocessing import Pool
from sklearn.metrics import confusion_matrix, ConfusionMatrixDisplay
import matplotlib.pyplot as plt


def extract_features(image_path):
    img = cv2.imread(image_path)

    if img is None:
        return None

    img = cv2.resize(img, (128, 128))

    hsv = cv2.cvtColor(img, cv2.COLOR_BGR2HSV)
    mean, std = cv2.meanStdDev(hsv)
    color_feats = np.concatenate([mean, std]).flatten()

    gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY)
    hist = cv2.calcHist([gray], [0], None, [256], [0, 256])
    hist = cv2.normalize(hist, hist).flatten()

    return np.concatenate([color_feats, hist[:128]])


def process_class(class_path):
    features = []

    for img_file in os.listdir(class_path)[:200]:
        img_path = os.path.join(class_path, img_file)

        if not os.path.isfile(img_path):
            continue

        feat = extract_features(img_path)

        if feat is not None:
            features.append(feat)

    return features


if __name__ == "__main__":
    path = kagglehub.dataset_download("emmarex/plantdisease")
    print(f"Dataset downloaded to: {path}")

    possible_paths = [
        os.path.join(path, "PlantVillage", "PlantVillage"),
        os.path.join(path, "PlantVillage"),
        path
    ]

    dataset_path = next(
        (p for p in possible_paths
         if os.path.isdir(p) and any(
             os.path.isdir(os.path.join(p, c))
             for c in os.listdir(p)
         )),
        None
    )

    if dataset_path is None:
        raise FileNotFoundError("PlantVillage dataset folder not found.")

    all_classes = sorted([
        c for c in os.listdir(dataset_path)
        if os.path.isdir(os.path.join(dataset_path, c))
    ])

    with Pool() as pool:
        results = list(tqdm(
            pool.imap(
                process_class,
                [os.path.join(dataset_path, c) for c in all_classes]
            ),
            total=len(all_classes)
        ))

    features_all = []
    labels_all = []
    classes_with_features = []

    for i, feats in enumerate(results):
        if feats:
            features_all.extend(feats)
            labels_all.extend([all_classes[i]] * len(feats))
            classes_with_features.append(all_classes[i])

    X = np.array(features_all, dtype=np.float32)
    y = np.array(labels_all)

    if len(X) == 0:
        raise ValueError("No images were processed. Check the dataset path.")

    print(f"Number of classes with images: {len(classes_with_features)}")
    print(f"Total samples: {len(y)}")

    X_train, X_test, y_train, y_test = train_test_split(
        X,
        y,
        test_size=0.2,
        stratify=y,
        random_state=42
    )

    scaler = StandardScaler()
    X_train = scaler.fit_transform(X_train)
    X_test = scaler.transform(X_test)

    pca = PCA(n_components=0.95, random_state=42)
    X_train = pca.fit_transform(X_train)
    X_test = pca.transform(X_test)

    print(
        f"Reduced from {X.shape[1]} to "
        f"{X_train.shape[1]} dimensions"
    )

    svm = SVC(
        kernel="rbf",
        C=5,
        gamma="scale",
        class_weight="balanced",
        cache_size=1000,
        random_state=42
    )

    svm.fit(X_train, y_train)

    y_pred = svm.predict(X_test)

    print(f"Accuracy: {accuracy_score(y_test, y_pred):.2f}")

    print(classification_report(
        y_test,
        y_pred,
        labels=classes_with_features,
        target_names=classes_with_features,
        zero_division=0
    ))

    cm = confusion_matrix(
        y_test,
        y_pred,
        labels=classes_with_features
    )

    disp = ConfusionMatrixDisplay(
        confusion_matrix=cm,
        display_labels=classes_with_features
    )

    fig, ax = plt.subplots(figsize=(15, 15))
    disp.plot(ax=ax, xticks_rotation=90, colorbar=False)
    plt.tight_layout()
    plt.show()

import pandas as pd
import numpy as np
import matplotlib.pyplot as plt
import seaborn as sns
from sklearn.preprocessing import StandardScaler
from sklearn.cluster import KMeans
from sklearn.mixture import GaussianMixture
from sklearn.metrics import silhouette_score
from sklearn.decomposition import PCA

url = "https://raw.githubusercontent.com/jbrownlee/Datasets/master/pima-indians-diabetes.data.csv"
cols = ["Pregnancies", "Glucose", "BloodPressure", "SkinThickness",
        "Insulin", "BMI", "DiabetesPedigreeFunction", "Age", "Outcome"]
df = pd.read_csv(url, header=None, names=cols)
print("Dataset shape:", df.shape)
print(df.head())

X = df.drop(columns=["Outcome"])
cols_zero = ["Glucose", "BloodPressure", "SkinThickness", "Insulin", "BMI"]
for c in cols_zero:
    X.loc[X[c] == 0, c] = X.loc[X[c] != 0, c].median()

X_scaled = StandardScaler().fit_transform(X)
n = 2

kmeans = KMeans(n_clusters=n, random_state=42)
kl = kmeans.fit_predict(X_scaled)
ks = silhouette_score(X_scaled, kl)
print(f"K-Means Silhouette Score: {ks:.3f}")

gmm = GaussianMixture(n_components=n, covariance_type="full", random_state=42)
gl = gmm.fit_predict(X_scaled)
gs = silhouette_score(X_scaled, gl)
print(f"EM (GMM) Silhouette Score: {gs:.3f}")

Xp = PCA(n_components=2).fit_transform(X_scaled)
fig, ax = plt.subplots(1, 2, figsize=(14, 6))

for a, labels, title, score in [
    (ax[0], kl, "K-Means", ks),
    (ax[1], gl, "EM (GMM)", gs)
]:
    sns.scatterplot(x=Xp[:, 0], y=Xp[:, 1], hue=labels,
                    palette="viridis", s=60, ax=a)
    a.set_title(f"{title} Clusters (Silhouette: {score:.3f})")
    a.set_xlabel("PCA 1")
    a.set_ylabel("PCA 2")

plt.tight_layout()
plt.show()

print("\nContingency table: K-Means vs Outcome")
print(pd.crosstab(kl, df["Outcome"]))
print("\nContingency table: EM vs Outcome")
print(pd.crosstab(gl, df["Outcome"]))

import numpy as np

import cv2
from sklearn.datasets import load_digits

from sklearn.preprocessing import OneHotEncoder
from sklearn.metrics import accuracy_score
from tensorflow.keras.models import Sequential
from tensorflow.keras.layers import Dense
from tensorflow.keras.optimizers import SGD

digits = load_digits()
X = digits.images.reshape(-1, 64)
y = digits.target

Y = OneHotEncoder().fit_transform(y.reshape(-1, 1)).toarray()
X_train, X_test, Y_train, Y_test = train_test_split(
    X, Y, test_size=0.2, random_state=42
)

model = Sequential([
    Dense(64, input_dim=64, activation="relu"),
    Dense(32, activation="relu"),
    Dense(10, activation="softmax")
])
model.compile(loss="categorical_crossentropy",
              optimizer=SGD(learning_rate=0.01), metrics=["accuracy"])
model.fit(X_train, Y_train, epochs=50, batch_size=16, verbose=0)

loss, accuracy = model.evaluate(X_test, Y_test, verbose=0)
print(f"Test Accuracy: {accuracy * 100:.2f}%")
model.save("digit_model.h5")

    print(f"Predicted Digit: {digit}")
    plt.imshow(img, cmap="gray")
    plt.title(f"Predicted Digit: {digit}")

import numpy as np
import pandas as pd
import cv2
import tensorflow as tf
import matplotlib.pyplot as plt
from sklearn.metrics import classification_report
from tensorflow.keras.models import Sequential
from tensorflow.keras.layers import Conv2D, MaxPooling2D, Flatten, Dense, Dropout
from tensorflow.keras.utils import to_categorical

def load_data():
    train = pd.read_csv("sign_mnist_train.csv")
    test = pd.read_csv("sign_mnist_test.csv")
    X_train = train.drop("label", axis=1).values.reshape(-1, 28, 28, 1) / 255.0
    X_test = test.drop("label", axis=1).values.reshape(-1, 28, 28, 1) / 255.0
    return X_train, to_categorical(train["label"], num_classes=25), X_test, to_categorical(test["label"], num_classes=25)

def build_model():
    model = Sequential([
        Conv2D(32, (3, 3), activation="relu", input_shape=(28, 28, 1)),
        MaxPooling2D(2, 2), Dropout(0.25),
        Conv2D(64, (3, 3), activation="relu"),
        MaxPooling2D(2, 2), Dropout(0.25),
        Flatten(), Dense(128, activation="relu"),
        Dropout(0.5), Dense(25, activation="softmax")
    ])
    model.compile(optimizer="adam", loss="categorical_crossentropy", metrics=["accuracy"])
    return model

def train_and_evaluate():
    X_train, y_train, X_test, y_test = load_data()
    model = build_model()
    history = model.fit(X_train, y_train, epochs=10, validation_data=(X_test, y_test))
    model.save("sign_language_model.h5")
    pred = model.predict(X_test)
    print(classification_report(np.argmax(y_test, axis=1), np.argmax(pred, axis=1)))
    plt.plot(history.history["accuracy"], label="Train")
    plt.plot(history.history["val_accuracy"], label="Val")
    plt.title("Model Accuracy")
    plt.xlabel("Epoch")
    plt.ylabel("Accuracy")
    plt.legend()
    plt.show()

def real_time_prediction():
    model = tf.keras.models.load_model("sign_language_model.h5")
    cap = cv2.VideoCapture(0)
    print("Press 'q' to quit.")
    while True:
        ret, frame = cap.read()
        if not ret:
            break
        roi = frame[100:300, 100:300]
        gray = cv2.cvtColor(roi, cv2.COLOR_BGR2GRAY)
        img = cv2.resize(gray, (28, 28)).reshape(1, 28, 28, 1) / 255.0
        letter = chr(np.argmax(model.predict(img, verbose=0)) + 65)
        if letter >= "J":
            letter = chr(ord(letter) + 1)
        cv2.rectangle(frame, (100, 100), (300, 300), (0, 255, 0), 2)
        cv2.putText(frame, f"Prediction: {letter}", (50, 50),
                    cv2.FONT_HERSHEY_SIMPLEX, 1.2, (0, 255, 0), 2)
        cv2.imshow("Sign Language Recognition", frame)
        if cv2.waitKey(1) & 0xFF == ord("q"):
            break
    cap.release()
    cv2.destroyAllWindows()

if __name__ == "__main__":
    import sys
    real_time_prediction() if len(sys.argv) > 1 and sys.argv[1] == "realtime" else train_and_evaluate()

import pandas as pd
import numpy as np
from scipy.sparse import coo_matrix, csr_matrix
import implicit
import warnings
warnings.filterwarnings("ignore")

books = pd.read_csv("Books.csv", encoding="utf-8-sig", on_bad_lines="skip", low_memory=False)
ratings = pd.read_csv("Ratings.csv", encoding="utf-8-sig", on_bad_lines="skip")
ratings = ratings[ratings["Book-Rating"] > 0]

users, isbns = sorted(ratings["User-ID"].unique()), sorted(ratings["ISBN"].unique())
user_idx = {u: i for i, u in enumerate(users)}
book_idx = {b: i for i, b in enumerate(isbns)}

matrix = coo_matrix((ratings["Book-Rating"],
    (ratings["User-ID"].map(user_idx), ratings["ISBN"].map(book_idx))),
    shape=(len(users), len(isbns))).tocsr()

model = implicit.als.AlternatingLeastSquares(
    factors=50, regularization=0.1, iterations=20, random_state=42)
model.fit(matrix.T)

def recommend_books_for_user(uid, n=5):
    if uid not in user_idx:
        return books.dropna(subset=["Book-Title", "Book-Author"])[
            ["Book-Title", "Book-Author"]].sample(
                min(n, len(books))).values.tolist()

    try:
        items, scores = model.recommend(
            user_idx[uid], csr_matrix(matrix[user_idx[uid]]),
            N=n, filter_already_liked_items=True)

        rec_isbns = [isbns[i] for i in items]
        results = books[books["ISBN"].isin(rec_isbns)][
            ["Book-Title", "Book-Author"]].drop_duplicates().values.tolist()

        if len(results) < n:
            extra = books[~books["ISBN"].isin(rec_isbns)].dropna(
                subset=["Book-Title", "Book-Author"])
            results += extra.sample(
                min(n - len(results), len(extra))) [
                ["Book-Title", "Book-Author"]].values.tolist()

        return results[:n]

    except Exception as e:
        print(f"Recommendation error: {e}")
        return books.dropna(subset=["Book-Title", "Book-Author"])[
            ["Book-Title", "Book-Author"]].sample(
                min(n, len(books))).values.tolist()

for uid in ratings["User-ID"].unique()[:5]:
    print(f"\nTop recommendations for User {uid}:")
    for i, (title, author) in enumerate(recommend_books_for_user(uid), 1):
        print(f"{i}. {title} by {author}")
