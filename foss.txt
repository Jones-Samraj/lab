const ImageGallery = {
    template: `
        <div class="gallery">

            <h1>Vue.js Image Gallery</h1>

            <!-- Event Handler -->
            <div class="controls">
                <input
                    type="text"
                    v-model="searchText"
                    placeholder="Search images..."
                >

                <br>

                <button @click="selectedCategory = 'All'">
                    All
                </button>

                <button @click="selectedCategory = 'Nature'">
                    Nature
                </button>

                <button @click="selectedCategory = 'Animals'">
                    Animals
                </button>
            </div>

            <!-- Dynamic Image Gallery -->
            <div class="image-grid">

                <div
                    class="image-card"
                    v-for="image in filteredImages"
                    :key="image.id"
                    @click="selectImage(image)"
                >

                    <img :src="image.url" :alt="image.title">

                    <h3>{{ image.title }}</h3>

                    <p>{{ image.category }}</p>

                </div>

            </div>

            <p v-if="filteredImages.length === 0">
                No images found.
            </p>

            <!-- Selected Image -->
            <div
                class="selected-image"
                v-if="selectedImage"
            >

                <h2>Selected Image</h2>

                <img
                    :src="selectedImage.url"
                    :alt="selectedImage.title"
                >

                <h3>{{ selectedImage.title }}</h3>

            </div>

        </div>
    `,

    data() {
        return {
            searchText: "",
            selectedCategory: "All",
            selectedImage: null,

            images: [
                {
                    id: 1,
                    title: "Mountain",
                    category: "Nature",
                    url: "https://images.unsplash.com/photo-1500534623283-312aade485b7"
                },
                {
                    id: 2,
                    title: "Forest",
                    category: "Nature",
                    url: "https://images.unsplash.com/photo-1448375240586-882707db888b"
                },
                {
                    id: 3,
                    title: "Lion",
                    category: "Animals",
                    url: "https://images.unsplash.com/photo-1549366021-9f761d450615"
                },
                {
                    id: 4,
                    title: "Elephant",
                    category: "Animals",
                    url: "https://images.unsplash.com/photo-1557050543-4d5f4e07ef46"
                },
                {
                    id: 5,
                    title: "Lake",
                    category: "Nature",
                    url: "https://images.unsplash.com/photo-1501785888041-af3ef285b470"
                },
                {
                    id: 6,

                }

            return this.images.filter(image => {

                    image.title
                        .toLowerCase()

        }
    }

auth.js


const router = require("express").Router();
const bcrypt = require("bcrypt");
const jwt = require("jsonwebtoken");
const User = require("./User");

const SECRET = "mysecretkey";

router.post("/register", async (req, res) => {
    const password = await bcrypt.hash(req.body.password, 10);

    const user = await User.create({
        username: req.body.username,
        email: req.body.email,
        password
    });

    res.json({ message: "Registered successfully", user });
});

router.post("/login", async (req, res) => {
    const user = await User.findOne({ email: req.body.email });

    if (!user || !(await bcrypt.compare(req.body.password, user.password)))
        return res.status(401).json({ message: "Invalid credentials" });

    const token = jwt.sign({ id: user._id }, SECRET);
    res.json({ token });
});

module.exports = router;


blog.js



const router = require("express").Router();
const jwt = require("jsonwebtoken");
const Blog = require("./Blog");

const SECRET = "mysecretkey";

function verifyToken(req, res, next) {
    try {
        const token = req.headers.authorization.split(" ")[1];
        req.user = jwt.verify(token, SECRET);
        next();
    } catch {
        res.status(401).json({ message: "Invalid token" });
    }
}

// GET all posts
router.get("/", async (req, res) => {
    res.json(await Blog.find());
});

// CREATE post
router.post("/", verifyToken, async (req, res) => {
    const post = await Blog.create({
        ...req.body,
        userId: req.user.id
    });
    res.json(post);
});

// UPDATE post using dynamic URL
router.put("/:id", verifyToken, async (req, res) => {
    const post = await Blog.findByIdAndUpdate(
        req.params.id,
        req.body,
        { new: true }
    );
    res.json(post);
});

// DELETE post
router.delete("/:id", verifyToken, async (req, res) => {
    await Blog.findByIdAndDelete(req.params.id);
    res.json({ message: "Post deleted" });
});

module.exports = router;

server.js
const express = require("express");
const http = require("http");
const { Server } = require("socket.io");
const redis = require("redis");

const app = express();
const server = http.createServer(app);
const io = new Server(server);

const pub = redis.createClient();
const sub = pub.duplicate();

app.use(express.json());
app.use(express.static("public"));

async function start() {
    await pub.connect();
    await sub.connect();

    await sub.subscribe("notifications", msg => {
        io.emit("notification", msg);
    });

    app.post("/follow", async (req, res) => {
        const { followerId, followedId } = req.body;
        const msg = `${followerId} followed ${followedId}`;

        await pub.publish("notifications", msg);
        res.json({ message: msg });
    });

    io.on("connection", () => console.log("User connected"));

    server.listen(3000, () =>
        console.log("Server running on port 3000")
    );
}

start();

import React, { useState } from "react";
import { useQuery, useMutation } from "@apollo/client";
import { GET_MSG_LIST } from "../graphql/queries";
import { ADD_MSG, UPDATE_MSG, DELETE_MSG } from "../graphql/mutations";

function TodoApp() {
    const { data } = useQuery(GET_MSG_LIST);
    const [addMsg] = useMutation(ADD_MSG);
    const [updateMsg] = useMutation(UPDATE_MSG);
    const [deleteMsg] = useMutation(DELETE_MSG);
    const [input, setInput] = useState("");

    const addTodo = () => {
        if (!input.trim()) return;

        addMsg({
            variables: { jobtodo: input },
            refetchQueries: [{ query: GET_MSG_LIST }]
        });
        setInput("");
    };

    return (
        <div>
            <h2>Task Management System</h2>

            <input
                value={input}
                onChange={e => setInput(e.target.value)}
                placeholder="Enter task"
            />

            <button onClick={addTodo}>Add</button>

            {data?.msglist.map(item => (
                <div key={item.id}>
                    {item.jobtodo}
                    <button onClick={() =>
                        updateMsg({
                            variables: {
                                id: item.id,
                                jobtodo: item.jobtodo
                            },
                            refetchQueries: [{ query: GET_MSG_LIST }]
                        })
                    }>
                        Update
                    </button>

                    <button onClick={() =>
                        deleteMsg({
                            variables: { id: item.id },
                            refetchQueries: [{ query: GET_MSG_LIST }]
                        })
                    }>
                        Delete
                    </button>
                </div>
            ))}
        </div>
    );
}

export default TodoApp;
